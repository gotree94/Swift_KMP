## 1. if 조건문

```swift
let score = 85

if score >= 90 {
    print("A")
} else if score >= 80 {
    print("B")
} else {
    print("C 이하")
}
```

### 조건 조합

```swift
let age = 20
let hasID = true

if age >= 18 && hasID {
    print("입장 가능")
}

if age < 18 || !hasID {
    print("입장 불가")
}
```

### 삼항 연산자

```swift
let max = 5 > 3 ? "5가 크다" : "3이 크다"
```

## 2. switch 구문

```swift
let grade = "B"

switch grade {
case "A":
    print("최고!")
case "B", "C":
    print("보통")
case let g where g == "D":     // 조건 필터 (where)
    print("노력 필요")
default:
    print("재시험")
}
```

- Swift의 `switch`는 **break를 쓰지 않아도 자동으로 탈출**합니다 (중요!).
- 모든 경우를 다루면 `default`를 생략해도 됩니다. 주로 `enum`에서 사용합니다.
- `fallthrough` 키워드로 다음 case로 넘길 수도 있습니다.

```swift
let action = "start"

switch action {
case "start":
    print("시작합니다")
    fallthrough   // 아래 case도 실행
case "init":
    print("초기화 중")
default:
    break
}
// 출력: "시작합니다" \n "초기화 중"
```

### 범위 switch

```swift
let count = 5

switch count {
case 1...3:
    print("적음")
case 4...10:
    print("보통")
default:
    print("많음")
}
```

## 3. guard (빠른 탈출)

`guard`는 조건이 거짓이면 **그 즉시 함수/반복을 종료**합니다. 조건이 참일 때만 이후 코드를 계속 실행합니다.
옵셔널 검사, 사전 조건 확인에 매우 널리 쓰입니다.

```swift
func checkAccess(user: String?, isAdmin: Bool) {
    guard let user = user else {
        print("사용자 정보가 없습니다. 종료.")
        return
    }
    guard isAdmin else {
        print("관리자가 아닙니다. 접근 거부.")
        return
    }
    print("\(user) 님, 환영합니다.")
}

checkAccess(user: "Minho", isAdmin: false)
checkAccess(user: nil, isAdmin: true)
```

> `if let` vs `guard let`:
> - if let: 옵셔널일 때 특정 분기만 처리 (블록 안에서만 사용)
> - guard let: nil이면 함수 전체에서 빠져나가야 할 때 (주로 함수 상단)

## 4. 반복문

### for-in

```swift
for i in 1...5 {
    print(i)                    // 1,2,3,4,5
}

for i in stride(from: 0, to: 10, by: 2) {   // 0,2,4,6,8
    print(i)
}

for i in stride(from: 10, through: 1, by: -2) {  // 10,8,6,4,2
    print(i)
}
```

### while / repeat-while

```swift
var count = 0
while count < 3 {
    count += 1
}
print(count)   // 3

var n = 0
repeat {                       // do-while과 유사, 최소 1회 실행
    n += 1
} while n < 3
print(n)   // 3
```

### break / continue

```swift
for i in 1...10 {
    if i % 2 == 0 { continue }    // 짝수면 건너뜀
    if i > 7 { break }            // 7 넘으면 전체 종료
    print(i)                      // 1 3 5 7
}
```

## 5. 함수 기초

```swift
// 기본 형태: 매개변수, 반환값
func greet(name: String) -> String {
    return "안녕, \(name)!"
}
print(greet(name: "Minho"))

// 반환값 없는 함수
func sayHi() {
    print("Hi!")
}

// 매개변수 여러 개
func add(a: Int, b: Int) -> Int {
    return a + b
}
print(add(a: 3, b: 5))   // 8
```

### 함수 호출시 매개변수 라벨

외부 이름과 내부 이름을 분리할 수 있습니다.

```swift
// 외부: to, 내부: person
func introduce(to person: String, from city: String) -> String {
    return "\(person)님은 \(city)에서 왔습니다."
}
print(introduce(to: "Minho", from: "Seoul"))
```

```swift
// 외부 라벨 생략 (_)
func multiply(_ a: Int, _ b: Int) -> Int {
    return a * b
}
print(multiply(3, 4))   // 12
```

### 기본값이 있는 매개변수

```swift
func order(coffee: String, size: String = "Medium", extra: Bool = false) -> String {
    return "\(size) \(coffee) 주문 완료 / 샷 추가: \(extra)"
}
print(order(coffee: "Latte"))                       // Medium Latte
print(order(coffee: "Americano", size: "Large"))    // Large Americano
```

## 6. 함수 심화

### 가변 매개변수 (Variadic)

```swift
func sum(_ numbers: Int...) -> Int {
    return numbers.reduce(0, +)
}
print(sum(1, 2, 3, 4))   // 10
```

### in-out 매개변수

함수 안에서 외부 변수 값을 직접 수정합니다.

```swift
func swap(_ a: inout Int, _ b: inout Int) {
    let temp = a
    a = b
    b = temp
}

var x = 10, y = 20
swap(&x, &y)
print(x, y)   // 20, 10
```

### 그리고 그 사이에 옵셔널 반환

```swift
func findName(id: Int, in dict: [Int: String]) -> String? {
    return dict[id]   // 없으면 nil
}

let users = [1: "Minho", 2: "Suji"]
if let name = findName(id: 2, in: users) {
    print("찾음: \(name)")
}
```

```swift
func average(of scores: [Int]) -> Int {
    return scores.reduce(0, +) / scores.count

func grade(of avg: Int) -> String {
    switch avg {
    case 90...:
        return "A"
    case 80..<90:
        return "B"
    case 70..<80:
        return "C"
    default:
        return "D"
    }
}

let scores = [92, 85, 70, 88]
```

```swift
import Foundation

func tax(annualIncome: Int) -> Int? {
    guard annualIncome > 0 else { return nil }
    let rate: Double
    if annualIncome <= 10_000_000 {
        rate = 0.06
    } else if annualIncome <= 50_000_000 {
        rate = 0.15
    } else {
        rate = 0.25
    }
    return Int(Double(annualIncome) * rate)
}

print("연봉 입력(원):", terminator: " ")
if let input = readLine(), let income = Int(input) {
    if let t = tax(annualIncome: income) {
        print("세금: \(t)원")
    } else {
        print("잘못된 입력")
    }
}
```
