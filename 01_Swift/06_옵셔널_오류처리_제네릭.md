## 1. 옵셔널 (Optional)

값이 **있거나 없을 수 있음**(nil)을 나타내는 타입입니다.

```swift
var name: String = "Minho"
// name = nil   // ❌ String에는 nil 불가

var nickname: String? = "민호"
nickname = nil   // ✅ Optional은 nil 가능

print(nickname)       // nil
print(nickname ?? "기본값")   // "기본값"
```

### Optional의 구조

`String?`은 사실 enum입니다: `.some("민호")` 또는 `.none`

```swift
let a: Int? = 10
print(a!)   // 10 (강제 언랩 - 비추천)

// 내부 구조 확인
switch a {
case .some(let value):
    print("값 있음: \(value)")
case .none:
    print("값 없음")
}
```

### if let 바인딩

```swift
let scores: [String: Int] = ["minho": 90]

// 안전한 언랩
if let score = scores["minho"] {
    print("점수: \(score)")
} else {
    print("없음")
}

// 여러 개 동시 바인딩
let first: String? = "A"
let second: Int? = nil
if let f = first, let s = second {
    print(f, s)
} else {
    print("하나라도 nil")   // 이쪽 실행
}
```

### guard let 바인딩

**가장 많이 쓰는 옵셔널 처리**입니다. 함수 전체에서 nil이면 조기 탈출.

```swift
func totalPrice(quantity: Int?, price: Int) -> Int? {
    guard let quantity = quantity else {
        print("수량 정보 없음")
        return nil
    }
    return quantity * price
}

print(totalPrice(quantity: 3, price: 500))   // 1500
print(totalPrice(quantity: nil, price: 500)) // nil + 로그
```

### nil 병합 연산자 / 옵셔널 체이닝

```swift
// nil 병합 ?? 
let value: Int? = nil
let safe = value ?? 100
print(safe)   // 100

// 옵셔널 체이닝: 중간에 nil이면 전체 nil
struct Address { var city: String }
struct User { var address: Address? }

let user = User(address: nil)
print(user.address?.city ?? "주소 없음")   // "주소 없음"

let user2 = User(address: Address(city: "Seoul"))
print(user2.address?.city ?? "주소 없음")  // "Seoul"
```

### 암시적 언랩 (실무에서 거의 안 씀)

```swift
var assumedName: String! = "Swift"   // 이론상 항상 값이 있다고 약속
print(assumedName)
```

## 2. 오류 처리 (Error Handling)

함수에서 오류가 나는 상황을 정의하고 처리합니다.

### 오류 타입 정의

```swift
enum BankError: Error {
    case insufficientFunds(balance: Int)
    case invalidAccount
}
```

### throw로 오류 발생

```swift
struct BankAccount {
    var balance: Int

    mutating func withdraw(_ amount: Int) throws {
        guard amount <= balance else {
            throw BankError.insufficientFunds(balance: balance)
        }
        balance -= amount
    }
}
```

### do-catch로 처리

```swift
var account = BankAccount(balance: 100)

do {
    try account.withdraw(500)
    print("출금 완료. 잔액: \(account.balance)")
} catch BankError.insufficientFunds(let balance) {
    print("잔액 부족. 현재 잔액: \(balance)")
} catch {
    print("기타 오류: \(error)")
}
```

### try? 와 try!

```swift
// try?: 실패하면 nil
let result = try? account.withdraw(10)
print(result ?? "실패")   // 10

// try!: 확실히 성공 보장 시 (크래시 위험)
// let forced = try! account.withdraw(99999)
```

## 3. 제네릭 (Generic)

타입을 파라미터처럼 받아 어떤 타입에도 동작하게 만듭니다.

### 제네릭 함수

```swift
func swapValues<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

var x = 1, y = 2
swapValues(&x, &y)
print(x, y)   // 2, 1

var s1 = "A", s2 = "B"
swapValues(&s1, &s2)
print(s1, s2)   // "B", "A"
```

### 제네릭 타입

```swift
struct Stack<T> {
    private var items: [T] = []

    mutating func push(_ item: T) {
        items.append(item)
    }

    mutating func pop() -> T? {
        return items.popLast()
    }
}

var intStack = Stack<Int>()
intStack.push(1)
intStack.push(2)
print(intStack.pop() ?? 0)   // 2

var strStack = Stack<String>()
strStack.push("Swift")
print(strStack.pop() ?? "")  // "Swift"
```

### 타입 제약 (Type Constraints)

특정 프로토콜을 준수하는 타입으로만 제한합니다.

```swift
func largest<T: Comparable>(in array: [T]) -> T? {
    return array.max()
}

print(largest(in: [3, 9, 7, 1]) ?? 0)      // 9
print(largest(in: ["a", "c", "b"]) ?? "")  // "c"
```

## 4. 더 알아두면 좋은 것

```swift
// Equatable/Hashable 자동 채택
struct Product: Equatable, Hashable {
    let id: Int
    let name: String
}
let p1 = Product(id: 1, name: "노트")
let p2 = Product(id: 1, name: "노트")
print(p1 == p2)   // true (구조체 자동 비교)

// Codable: JSON 변환의 기본 (다음 iOS 파트에서 사용)
```

<summary>실습 2 정답</summary>

```swift
enum WithdrawError: Error {
    case insufficient(balance: Int)
    case negativeAmount
    case overDailyLimit(limit: Int)
}

struct Account {
    var balance: Int
    var dailyWithdrawn = 0
    let dailyLimit = 500_000

    mutating func withdraw(_ amount: Int) throws {
        guard amount > 0 else { throw WithdrawError.negativeAmount }
        guard amount <= balance else { throw WithdrawError.insufficient(balance: balance) }
        guard dailyWithdrawn + amount <= dailyLimit else {
            throw WithdrawError.overDailyLimit(limit: dailyLimit)
        }
        balance -= amount
        dailyWithdrawn += amount
        print("출금 \(amount)원 성공. 잔액: \(balance)원")
    }
}

var account = Account(balance: 600_000)

do { try account.withdraw(700_000) }
catch WithdrawError.insufficient(let balance) { print("잔액 부족 (\(balance)원)") }
catch WithdrawError.negativeAmount { print("음수 금액") }
catch WithdrawError.overDailyLimit(let limit) { print("일일 한도 초과 (\(limit)원)") }
catch { print("기타 오류") }
```
</details>

<summary>실습 3 정답</summary>

```swift
struct CacheBox<T> {
    private var storage: [String: T] = [:]

    func get(_ key: String) -> T? {
        return storage[key]
    }

    mutating func set(_ key: String, _ value: T) {
        storage[key] = value
    }

    var count: Int { storage.count }
}

var intCache = CacheBox<Int>()
intCache.set("score", 95)
print(intCache.get("score") ?? 0)     // 95

var stringCache = CacheBox<String>()
stringCache.set("title", "Swift")
print(stringCache.get("title") ?? "")  // "Swift"

var listCache = CacheBox<[Int]>()
listCache.set("data", [1, 2, 3])
print(listCache.get("data") ?? [])     // [1, 2, 3]
```
