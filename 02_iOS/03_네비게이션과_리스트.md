# 2.3 네비게이션과 리스트

> List와 NavigationStack으로 여러 화면을 오가고, 데이터 목록을 표시하는 앱을 만든다.

---

## 학습 목표

- `List`로 동적 데이터 목록을 표시할 수 있다.
- `NavigationStack` + `NavigationLink`로 화면 전환을 구현할 수 있다.
- `ForEach`, `Identifiable`, `Section`을 이해한다.
- 호러패드(List) 삭제·이동(스와이프) 기능을 추가할 수 있다.

---

## 1. List로 데이터 목록 표시

```swift
struct FruitListView: View {
    let fruits = ["사과", "바나나", "오렌지"]

    var body: some View {
        List(fruits, id: \.self) { fruit in
            Text(fruit)
        }
    }
}
```

### ForEach와 Identifiable

더 좋은 방법은 각 항목이 고유 ID를 갖도록 **Identifiable** 프로토콜을 쓰는 것입니다.

```swift
struct Fruit: Identifiable {
    let id = UUID()          // 고유 식별자
    let name: String
    let emoji: String
}

struct FruitListView: View {
    let fruits = [
        Fruit(name: "사과", emoji: "🍎"),
        Fruit(name: "바나나", emoji: "🍌"),
        Fruit(name: "오렌지", emoji: "🍊")
    ]

    var body: some View {
        List(fruits) { fruit in      // id 불필요 (Identifiable)
            HStack {
                Text(fruit.emoji)
                Text(fruit.name)
            }
        }
    }
}
```

### Section으로 그룹화

```swift
List {
    Section("과일") {
        Text("사과")
        Text("바나나")
    }
    Section("채소") {
        Text("당근")
        Text("브로콜리")
    }
}
```

## 2. NavigationStack 화면 전환

### 기본 사용법

```swift
struct HomeView: View {
    var body: some View {
        NavigationStack {
            List {
                NavigationLink {
                    DetailView(title: "첫 번째 상세")
                } label: {
                    Text("첫 번째 상세로 이동")
                }
                NavigationLink {
                    DetailView(title: "두 번째 상세")
                } label: {
                    Text("두 번째 상세로 이동")
                }
            }
            .navigationTitle("메인 화면")
        }
    }
}

struct DetailView: View {
    let title: String

    var body: some View {
        Text(title)
            .font(.largeTitle)
            .navigationTitle(title)   // 상단 타이틀
            .navigationBarTitleDisplayMode(.inline)
    }
}
```

### 데이터와 함께 전달

`NavigationLink`의 값 가리기(value-based) 방식도 있습니다.

```swift
struct Book: Identifiable, Hashable {
    let id = UUID()
    let title: String
    let author: String
}

struct BookListView: View {
    let books = [
        Book(title: "어린왕자", author: "생텍쥐페리"),
        Book(title: "1984", author: "조지 오웰")
    ]

    var body: some View {
        NavigationStack {
            List(books) { book in
                NavigationLink(book.title) {
                    BookDetailView(book: book)
                }
            }
            .navigationTitle("도서 목록")
        }
    }
}

struct BookDetailView: View {
    let book: Book

    var body: some View {
        VStack(spacing: 12) {
            Text(book.title).font(.title)
            Text(book.author).foregroundStyle(.secondary)
        }
        .navigationTitle(book.title)
    }
}
```

### toolbar로 네비게이션 링크 추가

## 3. List 데이터 수정 (삭제·이동·검색)

### 삭제 (스와이프 + onDelete)

```swift
struct TaskListView: View {
    @State private var tasks = ["정리", "공부", "운동", "요리"]

    var body: some View {
        NavigationStack {
            List {
                ForEach(tasks, id: \.self) { task in
                    Text(task)
                }
                .onDelete { indexSet in       // 스와이프 삭제
                    tasks.remove(atOffsets: indexSet)
                }
                .onMove { from, to in         // 편집모드 순서 변경
                    tasks.move(fromOffsets: from, toOffset: to)
                }
            }
            .navigationTitle("할 일")
            .toolbar {
                EditButton()                  // 편집 모드 토글 (이동)
            }
        }
    }
}
```

### 검색 (searchable)

```swift
struct SearchableListView: View {
    let allFruits = ["사과", "바나나", "오렌지", "포도", "키위", "복숭아"]
    @State private var searchText = ""

    var filtered: [String] {
        if searchText.isEmpty {
            return allFruits
        }
        return allFruits.filter { $0.contains(searchText) }
    }

    var body: some View {
        NavigationStack {
            List(filtered, id: \.self) { name in
                Text(name)
            }
            .searchable(text: $searchText, prompt: "과일 검색")
            .navigationTitle("과일")
        }
    }
}
```

## 4. 실전 예제: 장바구니 앱

`CartApp`을 만들어 아래 전체 코드를 입력하고 실행해 보세요.

```swift
import SwiftUI

struct CartItem: Identifiable {
    let id = UUID()
    let name: String
    let price: Int
}

struct CartView: View {
    @State private var items = [
        CartItem(name: "아메리카노", price: 1500),
        CartItem(name: "쿠키", price: 2000),
        CartItem(name: "샌드위치", price: 4500)
    ]

    var body: some View {
        NavigationStack {
            List {
                ForEach(items) { item in
                    NavigationLink {
                        CartItemDetail(item: item)
                    } label: {
                        HStack {
                            Text(item.name)
                            Spacer()
                            Text("\(item.price)원")
                                .foregroundStyle(.secondary)
                        }
                    }
                }
                .onDelete { indexSet in items.remove(atOffsets: indexSet) }

                Section("합계") {
                    Text("총 \(items.reduce(0) { $0 + $1.price })원")
                        .font(.headline)
                }
            }
            .navigationTitle("장바구니")
            .toolbar { EditButton() }
        }
    }
}

struct CartItemDetail: View {
    let item: CartItem

    var body: some View {
        VStack(spacing: 12) {
            Image(systemName: item.name == "아메리카노" ? "cup.and.saucer.fill" : "fork.knife")
                .font(.system(size: 80))
                .foregroundStyle(.brown)
            Text(item.name).font(.title)
            Text("가격: \(item.price)원").foregroundStyle(.secondary)
        }
        .navigationTitle(item.name)
    }
}

#Preview {
    CartView()
}
```

## 🛠 실습: 음악 플레이리스트 앱

새 프로젝트 `PlaylistApp`을 만드세요.

**요구사항**

1. `Song` 구조체(Identifiable): `title: String`, `artist: String`, `minutes: Int`, `genre: String`
2. 플레이리스트 6곡을 리스트로 표시
3. 각 셀: 곡 제목 + 아티스트 (+ 오른쪽 시간)
4. 탭하면 상세 화면: 제목/아티스트/장르/시간 + 재생 버튼
5. `EditButton`으로 순서 변경, 스와이프 삭제 가능
6. `TextField` 없는 단순 검색 대신, `searchable`로 장르 검색(필터) 추가
7. 툴바에 "추가" 버튼 → 새 화면에서 TextField로 곡 추가(배열 append)

**실습 힌트**

- 장르 필터: `.filter { $0.genre.contains(searchText) || searchText.isEmpty }`
- 추가 화면에서 쓰기: `@State songTitle`, `@State artist`, 완료 시 `PlayerData.shared...` 보다는 바인딩 콜백으로.

```swift
// 부모 -> 자식으로 콜백 전달 패턴
NavigationLink {
    AddSongView { title, artist in
        songs.append(Song(title: title, artist: artist, ...))
    }
} label: {
    Image(systemName: "plus")
}
```
